```typescript
import Database from 'better-sqlite3';

export class LocalDatabase {
  private db: Database.Database;
  
  constructor(dbPath: string) {
    this.db = new Database(dbPath);
    this.initTables();
  }
  
  private initTables() {
    // 创建SQL存储表
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS sql_items (
        id TEXT PRIMARY KEY,
        title TEXT NOT NULL,
        content TEXT NOT NULL,
        description TEXT,
        category TEXT,
        tags TEXT,
        database_type TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        version INTEGER DEFAULT 1,
        is_favorite BOOLEAN DEFAULT 0,
        usage_count INTEGER DEFAULT 0
      )
    `);
    
    // 创建分类表
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS categories (
        id TEXT PRIMARY KEY,
        name TEXT NOT NULL,
        description TEXT,
        color TEXT,
        parent_id TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `);
  }
  
  // 保存SQL
  saveSQL(sqlItem: SQLItem): void {
    const stmt = this.db.prepare(`
      INSERT OR REPLACE INTO sql_items 
      (id, title, content, description, category, tags, database_type, version, is_favorite, usage_count)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `);
    
    stmt.run(
      sqlItem.id,
      sqlItem.title,
      sqlItem.content,
      sqlItem.description,
      sqlItem.category,
      JSON.stringify(sqlItem.tags),
      sqlItem.databaseType,
      sqlItem.version,
      sqlItem.isFavorite ? 1 : 0,
      sqlItem.usageCount
    );
  }
  
  // 查询SQL列表
  listSQLs(filters?: SQLFilters): SQLItem[] {
    let query = 'SELECT * FROM sql_items WHERE 1=1';
    const params: any[] = [];
    
    if (filters?.category) {
      query += ' AND category = ?';
      params.push(filters.category);
    }
    
    if (filters?.tags && filters.tags.length > 0) {
      query += ' AND tags LIKE ?';
      params.push(`%${filters.tags.join('%')}%`);
    }
    
    query += ' ORDER BY updated_at DESC';
    
    const stmt = this.db.prepare(query);
    const rows = stmt.all(...params);
    
    return rows.map(row => ({
      ...row,
      tags: JSON.parse(row.tags || '[]'),
      createdAt: new Date(row.created_at),
      updatedAt: new Date(row.updated_at)
    }));
  }
}
```

```typescript
import * as crypto from 'crypto';

export class EncryptionService {
  private algorithm = 'aes-256-cbc';
  private key: Buffer;
  
  constructor(secretKey: string) {
    // 使用环境变量或用户配置的密钥
    this.key = crypto.scryptSync(secretKey, 'salt', 32);
  }
  
  encrypt(text: string): string {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipher(this.algorithm, this.key);
    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    return iv.toString('hex') + ':' + encrypted;
  }
  
  decrypt(encryptedText: string): string {
    const textParts = encryptedText.split(':');
    const iv = Buffer.from(textParts.shift()!, 'hex');
    const encrypted = textParts.join(':');
    const decipher = crypto.createDecipher(this.algorithm, this.key);
    let decrypted = decipher.update(encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    return decrypted;
  }
}
```

```typescript
export class DatabaseConnectionManager {
  private connections = new Map<string, DatabaseConnection>();
  private connectionPools = new Map<string, any>();
  
  // 创建连接
  async createConnection(config: ConnectionConfig): Promise<string> {
    const connectionId = this.generateConnectionId();

          connection = await this.createMySQLConnection(config);

        case 'postgresql':
          connection = await this.createPostgreSQLConnection(config);

        config,
        connection,
        isActive: true,
        createdAt: new Date()
      });
      
      return connectionId;
    } catch (error) {
      throw new Error(`Failed to create connection: ${error.message}`);
    }
  }
  
  // MySQL连接
  private async createMySQLConnection(config: ConnectionConfig) {
    const mysql = require('mysql2/promise');
    return await mysql.createConnection({
      host: config.host,
      port: config.port,
      user: config.username,
      password: config.password,
      database: config.database,
      connectTimeout: 10000,
      acquireTimeout: 10000,
      timeout: 10000
    });
  }
  
  // PostgreSQL连接
  private async createPostgreSQLConnection(config: ConnectionConfig) {
    const { Client } = require('pg');
    const client = new Client({
      host: config.host,
      port: config.port,
      user: config.username,
      password: config.password,
      database: config.database,
      connectionTimeoutMillis: 10000
    });
    
    await client.connect();
    return client;
  }
  
  // 执行SQL
  async executeSQL(connectionId: string, sql: string): Promise<ExecutionResult> {
    const connection = this.connections.get(connectionId);
    if (!connection || !connection.isActive) {
      throw new Error('Connection not found or inactive');
    }
    
    try {
      const startTime = Date.now();
      let result: any;
      
      switch (connection.config.type) {
        case 'mysql':
          [result] = await connection.connection.execute(sql);
          break;
        case 'postgresql':
          result = await connection.connection.query(sql);
          break;
        case 'sqlite':
          result = connection.connection.prepare(sql).all();
          break;
      }
      
      const executionTime = Date.now() - startTime;
      
      return {
        success: true,
        data: result.rows || result,
        rowCount: result.affectedRows || result.rowCount || 0,
        executionTime,
        sql
      };
    } catch (error) {
      return {
        success: false,
        error: error.message,
        sql
      };
    }
  }
}
```

```typescript
export class SQLErrorLocator {
  // 定位错误位置
  locateError(sql: string, error: string): ErrorLocation {
    // 尝试从错误信息中提取行号和列号
    const lineMatch = error.match(/at line (\d+)/i);
    const columnMatch = error.match(/at column (\d+)/i);
    
    if (lineMatch && columnMatch) {
      return {
        line: parseInt(lineMatch[1]),
        column: parseInt(columnMatch[1]),
        message: error
      };
    }
    
    // 使用启发式方法定位错误
    return this.heuristicErrorLocation(sql, error);
  }
  
  // 启发式错误定位
  private heuristicErrorLocation(sql: string, error: string): ErrorLocation {
    const lines = sql.split('\n');
    
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      
      // 检查常见的SQL错误模式
      if (this.containsErrorPattern(line, error)) {
        return {
          line: i + 1,
          column: 1,
          message: error
        };
      }
    }
    
    return {
      line: 1,
      column: 1,
      message: error
    };
  }
}
```

```typescript
export class MySQLExecutionPlanAnalyzer {
  async analyzeExecutionPlan(connection: any, sql: string): Promise<ExecutionPlan> {
    try {
      // 获取执行计划
      const [rows] = await connection.execute(`EXPLAIN FORMAT=JSON ${sql}`);
      const plan = rows[0]['EXPLAIN'];
      
      return this.parseExecutionPlan(plan);
    } catch (error) {
      throw new Error(`Failed to analyze execution plan: ${error.message}`);
    }
  }
  
  private parseExecutionPlan(plan: any): ExecutionPlan {
    const analysis = {
      cost: 0,
      rows: 0,
      warnings: [],
      recommendations: []
    };
    
    // 分析执行计划
    this.analyzeNode(plan, analysis);
    
    // 生成优化建议
    analysis.recommendations = this.generateRecommendations(analysis);
    
    return analysis;
  }
  
  private analyzeNode(node: any, analysis: any) {
    if (node.cost_info) {
      analysis.cost += parseFloat(node.cost_info.query_cost);
    }
    
    if (node.rows_examined_per_scan) {
      analysis.rows += parseInt(node.rows_examined_per_scan);
    }
    
    // 检查潜在问题
    if (node.key === null && node.type === 'ALL') {
      analysis.warnings.push('Full table scan detected - consider adding indexes');
    }
    
    if (node.extra && node.extra.includes('Using filesort')) {
      analysis.warnings.push('Filesort detected - consider optimizing ORDER BY');
    }
    
    // 递归分析子节点
    if (node.children) {
      node.children.forEach((child: any) => this.analyzeNode(child, analysis));
    }
  }
}
```

### 6.1 缓存机制

#### 6.1.1 内存缓存
```typescript
export class CacheManager {
  private cache = new Map<string, CacheItem>();
  private maxSize = 1000;
  private ttl = 5 * 60 * 1000; // 5分钟
  
  // 获取缓存
  get<T>(key: string): T | null {
    const item = this.cache.get(key);
    
    if (!item) {
      return null;
    }
    
    // 检查过期
    if (Date.now() - item.timestamp > this.ttl) {
      this.cache.delete(key);
      return null;
    }
    
    return item.value;
  }
  
  // 设置缓存
  set<T>(key: string, value: T): void {
    // 检查缓存大小
    if (this.cache.size >= this.maxSize) {
      this.evictOldest();
    }
    
    this.cache.set(key, {
      value,
      timestamp: Date.now()
    });
  }
  
  // 清理过期项
  private evictOldest(): void {
    let oldestKey: string | null = null;
    let oldestTime = Date.now();
    
    for (const [key, item] of this.cache.entries()) {
      if (item.timestamp < oldestTime) {
        oldestTime = item.timestamp;
        oldestKey = key;
      }
    }
    
    if (oldestKey) {
      this.cache.delete(oldestKey);
    }
  }
}
```

#### 6.1.2 数据库查询缓存
```typescript
export class QueryCache {
  private cache = new CacheManager();
  
  // 缓存查询结果
  async getCachedResult<T>(
    sql: string,
    connectionId: string,
    fetchFn: () => Promise<T>
  ): Promise<T> {
    const cacheKey = this.generateCacheKey(sql, connectionId);
    
    // 尝试从缓存获取
    const cached = this.cache.get<T>(cacheKey);
    if (cached) {
      return cached;
    }
    
    // 执行查询
    const result = await fetchFn();
    
    // 缓存结果
    this.cache.set(cacheKey, result);
    
    return result;
  }
  
  // 生成缓存键
  private generateCacheKey(sql: string, connectionId: string): string {
    const normalizedSQL = this.normalizeSQL(sql);
    return `${connectionId}:${this.hash(normalizedSQL)}`;
  }
  
  // 标准化SQL
  private normalizeSQL(sql: string): string {
    return sql
      .replace(/\s+/g, ' ')
      .replace(/\s*([,()])\s*/g, '$1')
      .toLowerCase()
      .trim();
  }
}
```

```typescript
export class BackgroundTaskManager {
  private tasks = new Map<string, BackgroundTask>();
  
  // 启动后台任务
  async startTask<T>(
    taskId: string,
    taskFn: () => Promise<T>,
    onProgress?: (progress: number) => void
  ): Promise<string> {
    const task: BackgroundTask = {
      id: taskId,
      status: 'running',
      startTime: Date.now(),
      progress: 0
    };
    
    this.tasks.set(taskId, task);
    
    // 在后台执行任务
    setImmediate(async () => {
      try {
        await taskFn();
        task.status = 'completed';
        task.progress = 100;
      } catch (error) {
        task.status = 'failed';
        task.error = error.message;
      }
      
      task.endTime = Date.now();
    });
    
    return taskId;
  }
  
  // 获取任务状态
  getTaskStatus(taskId: string): BackgroundTask | null {
    return this.tasks.get(taskId) || null;
  }
  
  // 取消任务
  cancelTask(taskId: string): boolean {
    const task = this.tasks.get(taskId);
    if (task && task.status === 'running') {
      task.status = 'cancelled';
      return true;
    }
    return false;
  }
}
```
